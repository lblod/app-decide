* Fetch all the organizations withing the decide database

** Create .ttl file in the data/datadumps folder so it can be used as the downloadUrl of the Organizations distribution in the DCAT. 

#+begin_src sparql :format text/turtle :results file raw :file ../data/datadumps/20260826000000-organizations-new.ttl
  PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
  PREFIX besluit: <http://data.vlaanderen.be/ns/besluit#>
  PREFIX org: <http://www.w3.org/ns/org#>
  PREFIX mandaat: <http://data.vlaanderen.be/ns/mandaat#>
  PREFIX besluit: <http://data.vlaanderen.be/ns/besluit#>

  construct {
    ?organization ?p ?o .
  }
  where {
    ?organization a org:Organization  .
    ?organization ?p ?o .

    FILTER NOT EXISTS {
      VALUES ?classificatie {
        <http://data.vlaanderen.be/id/concept/BestuurseenheidClassificatieCode/36a82ba0-7ff1-4697-a9dd-2e94df73b721> # Autonoom Gemeentebedrijf
        <http://data.vlaanderen.be/id/concept/BestuurseenheidClassificatieCode/80310756-ce0a-4a1b-9b8e-7c01b6cc7a2d> # Autonoom Provinciebedrijf
      }
      ?organization besluit:classificatie ?classificatie . 
    }

    # No bestuursorganen in tijd
    FILTER NOT EXISTS {
      ?orgaanInTijd mandaat:isTijdspecialisatieVan ?organization .
    }

    # No AI generated organizations
    filter not exists {
      ?statement rdf:object ?organization .

      filter not exists {
        ?statement rdf:predicate <http://www.w3.org/2004/02/skos/core#exactMatch> .
      }
    }
  }
#+end_src
